cmake_minimum_required(VERSION 3.25)
project(vvk_loader_validation LANGUAGES C CXX)
find_package(VulkanHeaders CONFIG REQUIRED)
get_target_property(header_links Vulkan::Headers INTERFACE_LINK_LIBRARIES)
if(header_links)
    message(FATAL_ERROR "Vulkan::Headers unexpectedly links ${header_links}")
endif()
add_library(vvk-test-loader SHARED library.c)
add_library(vvk-test-missing-root SHARED library.c)
target_compile_definitions(vvk-test-missing-root PRIVATE VVK_MISSING_ROOT)
foreach(target vvk-test-loader vvk-test-missing-root)
    target_link_libraries(${target} PRIVATE Vulkan::Headers)
endforeach()
add_executable(headers-only headers.cpp)
target_link_libraries(headers-only PRIVATE Vulkan::Headers)
